% Hovering a module shows its docstring. `\pdfannot` is a pdfTeX primitive, so
% this needs no package: `pdfcomment` would be the idiomatic route and is absent
% from this TeX Live, along with `soul`, `soulpos`, `zref-abspage` and
% `marginnote`. The annotation is a zero-width, fully transparent /Square laid
% over the label's own box, which is what a reader hovers.
%
% Support is the reader's, not ours: Acrobat and most desktop readers show
% /Contents on hover; Chrome's built-in viewer and Firefox's pdf.js ignore it.
% The page is readable without a tooltip either way, which is why this is an
% addition and not the way a role is conveyed.
\newsavebox{\tipbox}
\newcommand{\tip}[2]{%
  \sbox{\tipbox}{#1}%
  \pdfannot width \wd\tipbox height \ht\tipbox depth \dp\tipbox {%
    /Subtype /Square /F 4 /CA 0 /IC [] /C []
    /BS << /W 0 >> /Contents (#2)%
  }%
  \usebox{\tipbox}%
}
